cmake_minimum_required(VERSION 3.16)
project(openarm_right_curl)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(OpenArmCAN REQUIRED)
add_library(right_curl_hw SHARED right_hardware.cpp)
target_compile_features(right_curl_hw PRIVATE cxx_std_17)
target_compile_options(right_curl_hw PRIVATE -Wall -Wextra)
ament_target_dependencies(right_curl_hw hardware_interface pluginlib rclcpp)
target_link_libraries(right_curl_hw OpenArmCAN::openarm_can)
pluginlib_export_plugin_description_file(hardware_interface plugin.xml)
install(TARGETS right_curl_hw LIBRARY DESTINATION lib)
ament_package()
