cmake_minimum_required(VERSION 3.16)
project(laiting_isaac_interface)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(laiting_isaac_hw SHARED sim_bridge.cpp)
target_compile_features(laiting_isaac_hw PRIVATE cxx_std_17)
ament_target_dependencies(laiting_isaac_hw hardware_interface pluginlib rclcpp sensor_msgs std_msgs)
pluginlib_export_plugin_description_file(hardware_interface plugin.xml)
install(TARGETS laiting_isaac_hw LIBRARY DESTINATION lib)
ament_package()
